cmake_minimum_required(VERSION 3.20)
project(lyrico_taglib LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(taglib_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
set(taglib_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/taglib")
if(NOT EXISTS "${taglib_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Initialize TagLib with: git submodule update --init --recursive")
endif()
add_subdirectory("${taglib_SOURCE_DIR}" "${taglib_BINARY_DIR}")
add_library(lyrico_taglib_bridge STATIC bridge.cpp)
target_include_directories(lyrico_taglib_bridge PRIVATE "${taglib_SOURCE_DIR}/taglib" "${taglib_SOURCE_DIR}/taglib/toolkit" "${taglib_BINARY_DIR}")
target_link_libraries(lyrico_taglib_bridge PRIVATE tag)
set_target_properties(lyrico_taglib_bridge PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS lyrico_taglib_bridge ARCHIVE DESTINATION lib RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
